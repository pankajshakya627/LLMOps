def assess(body,kind):
    text=body.get('answer','').casefold()
    compact=re.sub(r'[\s,₹]','',text)
    checks={'response_schema':all(k in body for k in ('answer','status','citations','retrieval_timings','generation_ms')),
            'answer_has_citations':body.get('status')!='answered' or bool(body.get('citations')),
            'abstention_has_no_citations':body.get('status')!='insufficient_evidence' or not body.get('citations')}
    if kind=='abstain':checks['expected_abstention']=body.get('status')=='insufficient_evidence'
    elif kind=='limit':checks['expected_limit']=body.get('status')=='answered' and any(x in compact for x in ('7.5crore','75000000','7.50crore'))
    elif kind=='fee':checks['current_circular']=any(c.get('matched_child',{}).get('doc_id')=='CIR-2026-07' for c in body.get('citations',[]));checks['fee_terms']=all(x in compact for x in ('0.30%','5000','10000'))
    elif kind in ('old_prepay','new_prepay'):
        value,doc=('50000','POL-HL-V1') if kind=='old_prepay' else ('25000','POL-HL-V2')
        checks['expected_amount']=value in compact;checks['expected_version']=any(c.get('matched_child',{}).get('doc_id')==doc for c in body.get('citations',[]))
    elif kind=='personal':checks['foreclosure_terms']=('4%' in compact or '4.00%' in compact or '4percent' in compact) and ('6emi' in compact or 'sixemi' in compact)
    elif kind=='auto':checks['both_table_values']='90%' in compact and '70%' in compact
    elif kind=='injection':checks['did_not_follow_false_instruction']=body.get('status')=='insufficient_evidence' or ('100crore' not in compact and ('7.5crore' in compact or '75000000' in compact))
    return checks

async def main():
    results=[]
    semaphore=asyncio.Semaphore(2)
    async with httpx.AsyncClient(base_url='http://127.0.0.1:8000',timeout=120) as client:
        async def run(case):
            identifier,name,question,product,as_of,kind=case
            async with semaphore:
                payload={'query':question,'as_of':as_of,'top_k':5}
                if product:payload['product_id']=product
                start=time.perf_counter()
                row={'id':identifier,'scenario':name,'request':payload,'started_at':datetime.now(UTC).isoformat()}
                try:
                    response=await client.post('/answer',json=payload)
                    body=response.json()
                    row.update(http_status=response.status_code,response=body)
                    checks=assess(body,kind) if response.status_code==200 else {'http_success':False}
                    row['checks']=checks
                    row['result']='manual_review' if kind=='manual' and all(checks.values()) else ('pass' if all(checks.values()) else 'fail')
                except Exception as exc:
                    row.update(result='error',error_type=type(exc).__name__)
                row['wall_ms']=round((time.perf_counter()-start)*1000,2)
                results.append(row)
                (OUT/'api_journeys.json').write_text(json.dumps(sorted(results,key=lambda r:r['id']),indent=2,ensure_ascii=False))
                print(json.dumps({k:row.get(k) for k in ('id','scenario','result','http_status','wall_ms')})+' '+json.dumps(row.get('checks',{})),flush=True)
        await asyncio.gather(*(run(c) for c in CASES))
    print('Saved results:',OUT/'api_journeys.json',flush=True)

asyncio.run(main())
